function doGet(e) {
  return handleRequest_(e, "GET");
}

function doPost(e) {
  return handleRequest_(e, "POST");
}

function handleRequest_(e, method) {
  var lock = LockService.getScriptLock();
  var gotLock = lock.tryLock(15000);
  try {
    var ss = getTargetSpreadsheet_();
    var dbSheet = ss.getSheetByName("DB_PENATAAN") || ss.insertSheet("DB_PENATAAN");
    var rekapSheet = ss.getSheetByName("REKAP_PENATAAN") || ss.insertSheet("REKAP_PENATAAN");

    var raw = String(dbSheet.getRange("A1").getValue() || "").trim();
    var state = getInitialState_();
    if (raw) {
      try {
        state = sanitizeState_(JSON.parse(raw));
      } catch (parseErr) {
        state = getInitialState_();
      }
    }

    var params = (e && e.parameter) ? e.parameter : {};
    var action = params.action ? String(params.action) : "get";
    var payload = {};

    if (e && e.postData && e.postData.contents) {
      var contents = String(e.postData.contents).trim();
      if (contents) {
        try {
          payload = JSON.parse(contents);
          if (payload && payload.action) {
            action = String(payload.action);
          }
        } catch (jsonErr) {
          if (params.payload) {
            try {
              payload = JSON.parse(String(params.payload));
            } catch (e2) {}
          }
        }
      }
    }

    if (!payload || typeof payload !== "object") {
      payload = {};
    }

    if (action === "save_judge" && payload.judge && state[payload.judge]) {
      var jKey = String(payload.judge);
      var rows = (payload.rows && typeof payload.rows === "object") ? payload.rows : {};
      Object.keys(rows).forEach(function(k) {
        var idx = parseInt(k, 10);
        if (!isNaN(idx) && idx >= 0 && idx < 16 && rows[k] && typeof rows[k] === "object") {
          var incomingTs = Number(rows[k].ts || 0);
          var currentTs = Number((state[jKey][idx] && state[jKey][idx].ts) || 0);
          if (incomingTs >= currentTs || incomingTs === 0) {
            state[jKey][idx] = {
              a1: rows[k].a1 !== undefined ? rows[k].a1 : "",
              a2: rows[k].a2 !== undefined ? rows[k].a2 : "",
              a3: rows[k].a3 !== undefined ? rows[k].a3 : "",
              a4: rows[k].a4 !== undefined ? rows[k].a4 : "",
              note: rows[k].note !== undefined ? rows[k].note : "",
              ts: incomingTs || new Date().getTime()
            };
          }
        }
      });
      state.updatedAt = new Date().getTime();
      saveStateToSheets_(dbSheet, rekapSheet, state);
    } else if (action === "merge_state" && payload.state) {
      state = mergeStates_(state, sanitizeState_(payload.state));
      state.updatedAt = new Date().getTime();
      saveStateToSheets_(dbSheet, rekapSheet, state);
    } else if (action === "update_settings") {
      if (Array.isArray(payload.classes) && payload.classes.length === 16) {
        state.classes = payload.classes.map(function(c, idx) {
          var s = String(c || "").trim();
          return s || getInitialState_().classes[idx];
        });
      }
      if (typeof payload.locked === "boolean") {
        state.locked = payload.locked;
      }
      state.updatedAt = new Date().getTime();
      saveStateToSheets_(dbSheet, rekapSheet, state);
    } else if (action === "import_state" && payload.state) {
      state = sanitizeState_(payload.state);
      state.updatedAt = new Date().getTime();
      saveStateToSheets_(dbSheet, rekapSheet, state);
    } else if (action === "reset_all") {
      ["j1", "j2", "j3"].forEach(function(jk) {
        for (var i = 0; i < 16; i++) {
          state[jk][i] = { a1: "", a2: "", a3: "", a4: "", note: "", ts: new Date().getTime() };
        }
      });
      state.updatedAt = new Date().getTime();
      saveStateToSheets_(dbSheet, rekapSheet, state);
    } else if (!raw) {
      saveStateToSheets_(dbSheet, rekapSheet, state);
    }

    return ContentService
      .createTextOutput(JSON.stringify({ ok: true, method: method, state: state }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: err && err.message ? err.message : String(err) }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    if (gotLock) {
      try { lock.releaseLock(); } catch (lockErr) {}
    }
  }
}

function getTargetSpreadsheet_() {
  var active = SpreadsheetApp.getActiveSpreadsheet();
  if (active) return active;
  var props = PropertiesService.getScriptProperties();
  var savedId = props.getProperty("SPREADSHEET_ID");
  if (savedId) {
    return SpreadsheetApp.openById(savedId);
  }
  throw new Error("Buka Apps Script langsung dari menu Extensions > Apps Script di file Google Sheet Anda.");
}

function saveStateToSheets_(dbSheet, rekapSheet, state) {
  dbSheet.getRange("A1").setValue(JSON.stringify(state));
  syncRekapSheet_(rekapSheet, state);
}

function getInitialState_() {
  var def = {
    classes: ["1A&24", "1B&2B", "1C&2C", "1D&2F", "1E&2E", "2D", "3A&4A", "3B&4B", "3C&4C", "3D&4D", "3E&4E", "5A&6A", "5B&6B", "5C&6C", "5D&6D", "5E&6E"],
    locked: false,
    j1: {},
    j2: {},
    j3: {},
    updatedAt: new Date().getTime()
  };
  ["j1", "j2", "j3"].forEach(function(jk) {
    for (var i = 0; i < 16; i++) {
      def[jk][i] = { a1: "", a2: "", a3: "", a4: "", note: "", ts: 0 };
    }
  });
  return def;
}

function sanitizeState_(obj) {
  var def = getInitialState_();
  if (!obj || typeof obj !== "object") return def;
  if (!Array.isArray(obj.classes) || obj.classes.length !== 16) {
    obj.classes = def.classes.slice();
  }
  if (typeof obj.locked !== "boolean") {
    obj.locked = false;
  }
  ["j1", "j2", "j3"].forEach(function(jk) {
    if (!obj[jk] || typeof obj[jk] !== "object") obj[jk] = {};
    for (var i = 0; i < 16; i++) {
      var r = obj[jk][i];
      if (!r || typeof r !== "object") {
        obj[jk][i] = { a1: "", a2: "", a3: "", a4: "", note: "", ts: 0 };
      } else {
        obj[jk][i] = {
          a1: r.a1 !== undefined ? r.a1 : "",
          a2: r.a2 !== undefined ? r.a2 : "",
          a3: r.a3 !== undefined ? r.a3 : "",
          a4: r.a4 !== undefined ? r.a4 : "",
          note: r.note !== undefined ? r.note : "",
          ts: Number(r.ts || 0)
        };
      }
    }
  });
  return obj;
}

function mergeStates_(baseObj, incObj) {
  var out = sanitizeState_(baseObj);
  var inc = sanitizeState_(incObj);
  if (inc.classes && inc.classes.length === 16) out.classes = inc.classes.slice();
  if (typeof inc.locked === "boolean") out.locked = inc.locked;
  ["j1", "j2", "j3"].forEach(function(jk) {
    for (var i = 0; i < 16; i++) {
      var bRow = out[jk][i];
      var iRow = inc[jk][i];
      var hasIncVal = (
        String(iRow.a1) !== "" ||
        String(iRow.a2) !== "" ||
        String(iRow.a3) !== "" ||
        String(iRow.a4) !== "" ||
        String(iRow.note) !== ""
      );
      if (hasIncVal && Number(iRow.ts || 0) >= Number(bRow.ts || 0)) {
        out[jk][i] = iRow;
      }
    }
  });
  return out;
}

function computeJudgeScore_(r) {
  if (!r) return 0;
  var n1 = parseFloat(r.a1), n2 = parseFloat(r.a2), n3 = parseFloat(r.a3), n4 = parseFloat(r.a4);
  if (isNaN(n1) && isNaN(n2) && isNaN(n3) && isNaN(n4)) return 0;
  var a1 = isNaN(n1) ? 0 : n1;
  var a2 = isNaN(n2) ? 0 : n2;
  var a3 = isNaN(n3) ? 0 : n3;
  var a4 = isNaN(n4) ? 0 : n4;
  return ((((a1 * 0.30) + (a2 * 0.30) + (a3 * 0.20) + (a4 * 0.20)) / 4) * 100);
}

function syncRekapSheet_(sheet, state) {
  var rows = [[
    "No", "Kelas",
    "J1_TataLetak", "J1_Administrasi", "J1_CahayaUdara", "J1_Estetika", "J1_Total",
    "J2_TataLetak", "J2_Administrasi", "J2_CahayaUdara", "J2_Estetika", "J2_Total",
    "J3_TataLetak", "J3_Administrasi", "J3_CahayaUdara", "J3_Estetika", "J3_Total",
    "Nilai_Akhir", "Catatan_Juri"
  ]];
  for (var i = 0; i < 16; i++) {
    var j1 = state.j1[i] || {};
    var j2 = state.j2[i] || {};
    var j3 = state.j3[i] || {};
    var t1 = Number(computeJudgeScore_(j1).toFixed(2));
    var t2 = Number(computeJudgeScore_(j2).toFixed(2));
    var t3 = Number(computeJudgeScore_(j3).toFixed(2));
    var akhir = Number(((t1 + t2 + t3) / 3).toFixed(2));
    var catatan = [
      j1.note ? "J1: " + j1.note : "",
      j2.note ? "J2: " + j2.note : "",
      j3.note ? "J3: " + j3.note : ""
    ].filter(Boolean).join(" | ");
    rows.push([
      i + 1,
      state.classes[i],
      j1.a1 || "", j1.a2 || "", j1.a3 || "", j1.a4 || "", t1,
      j2.a1 || "", j2.a2 || "", j2.a3 || "", j2.a4 || "", t2,
      j3.a1 || "", j3.a2 || "", j3.a3 || "", j3.a4 || "", t3,
      akhir,
      catatan
    ]);
  }
  sheet.clearContents();
  sheet.getRange(1, 1, rows.length, rows[0].length).setValues(rows);
}
